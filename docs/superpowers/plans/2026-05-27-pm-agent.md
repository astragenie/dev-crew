# PM Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `loop:pm` agent that replaces regex triage with LLM-powered product management — scoring, demand validation, scope challenge, and autonomous_safe computation.

**Architecture:** New `pm-scorer.mjs` module for deterministic scoring logic. New `pm-artifact-writer.mjs` for PM review artifacts. Agent prompt (`agents/pm.md`) embeds three gstack-ported frameworks. Three skills wire the agent into the loop lifecycle. Existing `backlog-triage.mjs` kept as fallback; `backlog-parser.mjs` enhanced to sort by composite priority.

**Tech Stack:** ESM Node.js (.mjs), node:test, existing frontmatter parser. No new dependencies.

**Important:** The frontmatter parser is flat key-value only (no nested YAML objects). PM scores stored as flat `pm_*` frontmatter keys. Narrative stored in `## PM Review` body section (same pattern as grades use `## Scores`).

```javascript
export const PM_DIMENSIONS = [
  "customer_impact",
  "effort_estimate",
  "strategic_alignment",
  "technical_risk",
  "dependency_depth"
];

export function validatePmScores(scores) {
  const errors = [];
  for (const dim of PM_DIMENSIONS) {
    const v = scores[dim];
    if (v === undefined || v === null) {
      errors.push(`missing dimension: ${dim}`);
      continue;
    }
    if (typeof v !== "number" || Number.isNaN(v)) {
      errors.push(`${dim} must be a number, got ${typeof v}`);
      continue;
    }
    if (v < 0 || v > 1) {
      errors.push(`${dim} = ${v} out of range [0, 1]`);
    }
  }
  return errors;
}

export function computeCompositePriority(scores, category = "feature") {
  const { customer_impact, effort_estimate, strategic_alignment, technical_risk } = scores;
  if (customer_impact >= 0.8 && (category === "bugfix" || technical_risk >= 0.8)) return "P0";
  if (
    (customer_impact >= 0.7 && strategic_alignment >= 0.6) ||
    (strategic_alignment >= 0.8 && effort_estimate <= 0.4)
  ) {
    return "P1";
  }
  if (effort_estimate >= 0.7 && customer_impact <= 0.3) return "P3";
  return "P2";
}

export function computeAutonomousSafe(scores, context = {}) {
  const { technical_risk, effort_estimate, dependency_depth } = scores;
  const { hasSpec = false, hasAcceptanceCriteria = false } = context;
  if (technical_risk > 0.6) return false;
  if (effort_estimate > 0.6) return false;
  if (dependency_depth > 0.6) return false;
  if (!hasSpec && !hasAcceptanceCriteria) return false;
  return true;
}

export function averagePmScore(scores) {
  const values = PM_DIMENSIONS.map((d) => scores[d]).filter((v) => typeof v === "number");
  if (values.length === 0) return null;
  return values.reduce((a, b) => a + b, 0) / values.length;
}

export function parsePmScoresFromFrontmatter(frontmatter) {
  const scores = {};
  for (const dim of PM_DIMENSIONS) {
    const key = `pm_${dim}`;
    if (frontmatter[key] !== undefined && frontmatter[key] !== null) {
      scores[dim] = Number(frontmatter[key]);
    }
  }
  return scores;
}
```

```javascript
import fs from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";

import { serializeFrontmatter } from "./frontmatter.mjs";
import { PM_DIMENSIONS } from "./pm-scorer.mjs";
import { pathExists } from "./paths.mjs";

const scriptDir = path.dirname(fileURLToPath(import.meta.url));
const templateDir = path.resolve(scriptDir, "..", "..", "templates");

function today() {
  return new Date().toISOString().slice(0, 10);
}

export async function writePmReviewArtifact(repoPath, {
  featureId,
  scores = {},
  compositePriority = "P2",
  autonomousSafe = false,
  demandEvidence = null,
  scopeNotes = null,
  risks = [],
  reasoning = null
}) {
  if (!featureId) throw new Error("writePmReviewArtifact requires featureId");
  const dir = path.join(repoPath, ".claude", "artifacts", "loop", "pm-reviews");
  await fs.mkdir(dir, { recursive: true });

  const tpl = await fs.readFile(path.join(templateDir, "pm-review.md"), "utf8");
  const id = `PM-REVIEW-${featureId}`;

  const frontmatter = {
    id,
    feature: featureId,
    reviewed_at: today()
  };
  for (const dim of PM_DIMENSIONS) {
    frontmatter[`pm_${dim}`] = scores[dim] ?? 0;
  }
  frontmatter.composite_priority = compositePriority;
  frontmatter.autonomous_safe = autonomousSafe;

  const scoreLine = (dim) =>
    `- ${dim}: ${(scores[dim] ?? 0).toFixed(2)}`;

  const body = [
    `# PM Review — ${featureId}`,
    "",
    "## Demand Assessment",
    "",
    demandEvidence ? `- **Evidence:** ${demandEvidence}` : "- **Evidence:** none",
    "",
    "## Scope Challenge",
    "",
    scopeNotes ? `- **Scope:** ${scopeNotes}` : "- **Scope:** not assessed",
    "",
    "## Risk Radar",
    "",
    ...(risks.length > 0 ? risks.map((r) => `- ${r}`) : ["- none identified"]),
    "",
    "## Scores",
    "",
    ...PM_DIMENSIONS.map(scoreLine),
    "",
    "## Priority Derivation",
    "",
    `composite_priority: ${compositePriority}`,
    `autonomous_safe: ${autonomousSafe}`,
    reasoning ? `reasoning: ${reasoning}` : "reasoning: not provided",
    ""
  ].join("\n");

  const target = path.join(dir, `${featureId}-pm-review.md`);
  await fs.writeFile(target, serializeFrontmatter(frontmatter, body));
  return { artifactPath: target, id };
}
```

```javascript
export async function writePmReview(filePath, {
  scores = {},
  compositePriority = "P2",
  autonomousSafe = false,
  demandEvidence = null,
  scopeNotes = null,
  risks = []
}) {
  const { PM_DIMENSIONS } = await import("./pm-scorer.mjs");
  const updates = {};
  for (const dim of PM_DIMENSIONS) {
    if (scores[dim] !== undefined) updates[`pm_${dim}`] = scores[dim];
  }
  updates.pm_composite_priority = compositePriority;
  updates.pm_autonomous_safe = autonomousSafe;
  if (demandEvidence !== null) updates.pm_demand_evidence = demandEvidence;
  if (scopeNotes !== null) updates.pm_scope_notes = scopeNotes;
  if (risks.length > 0) updates.pm_risks = risks;
  updates.pm_reviewed_at = nowIsoDate();
  const content = await fs.readFile(filePath, "utf8");
```

```markdown
# PM Agent — loop:pm

You are the product manager for this repository. Your job is to evaluate features in the backlog, score them across five dimensions, and produce a priority + autonomous_safe determination.

## Your output contract

For each FEAT you review, produce a JSON block:

\`\`\`json
{
  "featureId": "FEAT-NNN",
  "scores": {
    "customer_impact": 0.00,
    "effort_estimate": 0.00,
    "strategic_alignment": 0.00,
    "technical_risk": 0.00,
    "dependency_depth": 0.00
  },
  "compositePriority": "P2",
  "autonomousSafe": false,
  "demandEvidence": "string or none",
  "scopeNotes": "string",
  "risks": ["string"],
  "reasoning": "one-line summary"
}
\`\`\`

## Scoring dimensions (0.0 to 1.0, higher = more)

| Dimension | 0.0 | 1.0 |
|---|---|---|
| customer_impact | Nice-to-have, no evidence | Critical pain, strong evidence |
| effort_estimate | Trivial, single-file change | Multi-slice epic, cross-service |
| strategic_alignment | Tangential to mission | Core mission, roadmap centerpiece |
| technical_risk | Safe, well-understood patterns | Unknown territory, new deps |
| dependency_depth | Fully independent | Blocked by 3+ unresolved items |

## Framework 1: Demand Assessment

For each FEAT, answer:
- **Who wants this?** Name a specific user, persona, or internal stakeholder. Not "users would like..."
- **Evidence?** Search for: grade lessons mentioning this area, spec mandates, bug patterns. Quote what you find. If nothing, write `demand_evidence: "none"` and score customer_impact <= 0.3.
- **Current workaround?** How do people solve this today? If the workaround is tolerable, impact drops.
- **Frequency?** Daily pain or edge case?

## Framework 2: Scope Challenge

- **Smallest deliverable?** What is the minimum that delivers the core value?
- **What would you cut?** If you had to halve the scope, what stays?
- **Overlap?** Does this duplicate existing capability? Check existing FEATs in done/ and in-progress/.
- **AC clarity?** Read the feature body. Are acceptance criteria clear and testable? Vague AC = `autonomousSafe: false`.

## Framework 3: Risk Radar

- **Technical unknowns?** New technology, external API, database migration, schema change.
- **Dependency chain?** Check `depends_on` frontmatter. Check other FEATs' bodies for references to this feature's domain.
- **Grade history?** Read recent grades under `.claude/artifacts/loop/grades/`. Did past slices in this area score low on reliability, security, or test_confidence?
- **Blast radius?** Estimate how many files/services this touches.

## Anti-hallucination rules

- Quote evidence from files you actually read. Include the file path.
- Write `demand_evidence: "none"` when you have no backing. Never fabricate demand.
- If you cannot assess a dimension, score it 0.5 and note "insufficient data" in reasoning.
- Do not invent stakeholder names or user quotes.

## How to gather context

1. Read the FEAT file body and frontmatter
2. If `derived_from` is set, read the linked SPEC file
3. Read recent grades: `ls .claude/artifacts/loop/grades/` — scan lessons/surprises for mentions of this feature's domain
4. Read recent decisions: `ls .claude/artifacts/loop/decisions/` — check for prior architectural choices affecting this area
5. Check dependency graph: grep for this FEAT's id across all backlog files
6. Read `productDescription` and `productName` from `.claude/loop.json` for strategic alignment context
```

```markdown
# PM Triage

Triage all pending features in the backlog using the PM agent.

## Steps

1. List pending features: run the equivalent of `listBacklog` filtered to `pending/` state.
2. For each pending FEAT:
   a. Read the FEAT file (frontmatter + body).
   b. If `derived_from` is set, read the linked SPEC.
   c. Scan recent grades for demand evidence (`.claude/artifacts/loop/grades/`).
   d. Apply the three PM frameworks (Demand Assessment, Scope Challenge, Risk Radar).
   e. Score all 5 dimensions (0.0-1.0).
   f. Compute `compositePriority` using the derivation rules.
   g. Compute `autonomousSafe` using the safety rules.
3. For each scored FEAT, update its frontmatter with `pm_*` fields by running:
   ```
   node scripts/loop.mjs backlog pm-write --id FEAT-NNN --scores '{"customer_impact":0.8,...}' --priority P1 --autonomous-safe true --evidence "..." --reasoning "..."
   ```
4. Move each FEAT from `pending/` to `triaged/` via the standard triage flow.
5. Write a PM review artifact for each FEAT to `.claude/artifacts/loop/pm-reviews/`.
6. Return a summary table:

| FEAT | Priority | Safe? | Reasoning |
|------|----------|-------|-----------|
| FEAT-NNN | P1 | yes | one-line |

## When this runs

- Automatically: loop orchestrator dispatches before Cross-Slice Continuation
- Manually: user invokes `/loop:pm-triage`
```

```markdown
# PM Decompose — Spec to Scored Features

Split a SPEC into FEAT files with PM scores pre-filled.

## Usage

`/loop:pm-decompose --id SPEC-NNN`

## Steps

1. Read the SPEC file body and frontmatter.
2. Apply Scope Challenge framework to identify natural feature boundaries:
   - What are the independent deliverables?
   - What is the dependency order?
   - Which features can be autonomous vs. need human review?
3. For each proposed feature:
   a. Draft the FEAT title and description.
   b. Pre-score all 5 PM dimensions based on the spec context.
   c. Compute composite priority and autonomous_safe.
   d. Identify dependencies between the proposed features.
4. Present the decomposition to the user (or log if autonomous):
   - Feature list with scores and priorities
   - Dependency graph
   - Which features are autonomous_safe
5. Create FEAT files via `/loop:backlog-add` with PM scores pre-filled.
6. Update the SPEC's `derived_features` list.
7. Write PM review artifacts for each new FEAT.
```

### Task 8: Integration Test — Full PM Triage Flow

**Files:**
- Create: `tests/pm-triage-flow.test.mjs`

- [ ] **Step 1: Write integration test**

```javascript
import fs from "node:fs/promises";
import os from "node:os";
import path from "node:path";
import test from "node:test";
import assert from "node:assert/strict";

import { addFeature } from "../scripts/lib/backlog-writer.mjs";
import { listBacklog, filterBacklog } from "../scripts/lib/backlog-parser.mjs";
import { parseFrontmatter } from "../scripts/lib/frontmatter.mjs";
import {
  computeCompositePriority,
  computeAutonomousSafe,
  PM_DIMENSIONS,
  validatePmScores
} from "../scripts/lib/pm-scorer.mjs";
import { writePmReviewArtifact } from "../scripts/lib/pm-artifact-writer.mjs";

async function tmpRepo(prefix) {
  return fs.mkdtemp(path.join(os.tmpdir(), prefix));
}

test("full PM triage flow: add feature, score, write artifact, read back", async () => {
  const repo = await tmpRepo("al-pm-flow-");

  const feat = await addFeature(repo, {
    title: "Add pagination to ingestion API",
    body: "## Description\n\n10k+ jobs per tenant exceed payload limit.\n\n## Acceptance hints\n\n- API returns paginated results\n- Dashboard shows page controls\n",
    category: "feature"
  });

  const scores = {
    customer_impact: 0.8,
    effort_estimate: 0.35,
    strategic_alignment: 0.9,
    technical_risk: 0.3,
    dependency_depth: 0.1
  };

  const errors = validatePmScores(scores);
  assert.equal(errors.length, 0);

  const priority = computeCompositePriority(scores, "feature");
  assert.equal(priority, "P1");

  const safe = computeAutonomousSafe(scores, {
    hasSpec: false,
    hasAcceptanceCriteria: true
  });
  assert.equal(safe, true);

  const artifact = await writePmReviewArtifact(repo, {
    featureId: feat.id,
    scores,
    compositePriority: priority,
    autonomousSafe: safe,
    demandEvidence: "10k+ jobs payload limit hit in production",
    reasoning: "High impact, low effort, core API improvement"
  });

  assert.ok(await fs.access(artifact.artifactPath).then(() => true).catch(() => false));

  const artifactContent = await fs.readFile(artifact.artifactPath, "utf8");
  const { frontmatter: artifactFm } = parseFrontmatter(artifactContent);
  assert.equal(artifactFm.pm_customer_impact, 0.8);
  assert.equal(artifactFm.composite_priority, "P1");
  assert.equal(artifactFm.autonomous_safe, true);
});

test("validatePmScores catches missing dimensions", () => {
  const partial = { customer_impact: 0.5, effort_estimate: 0.3 };
  const errors = validatePmScores(partial);
  assert.equal(errors.length, 3);
  assert.ok(errors.some((e) => e.includes("strategic_alignment")));
});

test("validatePmScores catches out-of-range values", () => {
  const bad = {
    customer_impact: 1.5,
    effort_estimate: -0.1,
    strategic_alignment: 0.5,
    technical_risk: 0.5,
    dependency_depth: 0.5
  };
  const errors = validatePmScores(bad);
  assert.equal(errors.length, 2);
});
```
