function avg(values: unknown[]) {
  const nums = values.filter((v): v is number => typeof v === "number");
  if (nums.length === 0) return null;
  return Number((nums.reduce((a, b) => a + b, 0) / nums.length).toFixed(3));
}

// Parse frontmatter from markdown to extract metadata.
function parseFrontmatter(text: string) {
  if (!text.startsWith("---")) return { fm: null, body: text };
  const end = text.indexOf("\n---", 3);
  if (end < 0) return { fm: null, body: text };
  const block = text.slice(3, end).trim();
  const body = text.slice(end + 4);
  const fm: Record<string, unknown> = {};
  for (const line of block.split(/\r?\n/)) {
    const m = line.match(/^([\w_]+):\s*(.*)$/);
    if (!m) continue;
    const raw = (m[2] ?? "").trim();
    // Try to parse as number (for usd field)
    const v: string | number = /^\d+(\.\d+)?$/.test(raw) ? Number(raw) : raw;
    fm[m[1]!] = v;
  }
  return { fm, body };
}

// Load cost report USD values from the most recent cost reports.
// Returns array of USD values for cost trajectory calculation.
async function loadCostReportUSDValues(repoPath: string, limit = 20) {
  const costDirParts = [".claude", "artifacts", "crew", "cost"];
  const legacyDirParts = [".claude", "artifacts", "crew", "runs"];

  const dirs = [path.join(repoPath, ...costDirParts), path.join(repoPath, ...legacyDirParts)];

  const files: any[] = [];
  for (const dir of dirs) {
    let entries;
    try {
      entries = await fs.readdir(dir);
    } catch {
      continue;
    }
    for (const e of entries) {
      if (/-cost-report-.+\.md$/.test(e)) files.push({ dir, name: e });
    }
  }

  // Sort by basename desc (newest first via timestamp prefix)
  files.sort((a, b) => b.name.localeCompare(a.name));
  const top = files.slice(0, limit);

  const usdValues: any[] = [];
  for (const { dir, name } of top) {
    const f = path.join(dir, name);
    try {
      const text = await fs.readFile(f, "utf8");
      const { fm } = parseFrontmatter(text);
      if (fm && typeof fm.usd === "number") {
        usdValues.push(fm.usd);
      }
    } catch {
      /* skip unreadable files silently */
    }
  }

  return usdValues;
}

async function collectSnapshotData(
  repoPath: string,
  config: Record<string, any> | null
): Promise<{
  backlog: Record<string, any>[];
  specs: Record<string, any>[];
  grades: Record<string, any>[];
  decisions: Record<string, any>[];
  iterationStatus: Record<string, any> | null;
  costReportUSDValues: number[];
}> {
  const [backlog, specs, grades, decisions, iterationStatus, costReportUSDValues] =
    await Promise.all([
      listBacklog(repoPath, { config }),
      listSpecs(repoPath, { config }),
      listGrades(repoPath, { config }),
      listDecisions(repoPath, { config }),
      getIterationStatus(repoPath),
      loadCostReportUSDValues(repoPath)

}

function buildSnapshotLines(
  data: Awaited<ReturnType<typeof collectSnapshotData>>,
  today: string
): string[] {
  const { backlog, specs, grades, decisions, iterationStatus, costReportUSDValues } = data;
  const inFlight = filterBacklog(backlog, { status: "in-progress" });
  // Exclude deferred items from "next to slice" — mirrors the auto walker's
  // pre-branch filter (walker-gates.mts: `entry.frontmatter.deferred === true`
  // → skip). Deferred FEATs are still counted in Triaged: N (renderBacklogState);
  // only the next-to-slice candidate list drops them so the two views agree.
  const triagedTop = filterBacklog(backlog, { status: "triaged" })
    .filter((f: Record<string, any>) => f.frontmatter.deferred !== true)
    .sort(comparePickOrder)
    .slice(0, 5);
  const specsByStatus = {
    draft: specs.filter((s: Record<string, any>) => s.frontmatter.status === "draft").length,
    approved: specs.filter((s: Record<string, any>) => s.frontmatter.status === "approved").length,
    decomposed: specs.filter((s: Record<string, any>) => s.frontmatter.status === "decomposed")
      .length,
    satisfied: specs.filter((s: Record<string, any>) => s.frontmatter.status === "satisfied").length
  };
  const recentGrades = grades.slice(-5);
  const recentScoreAverages: Record<string, number | null> = {};
  for (const dim of SCORE_DIMENSIONS) {
    recentScoreAverages[dim] = avg(recentGrades.map((g: Record<string, any>) => g.scores?.[dim]));
  }
  const recentDecisions = decisions.slice(-5);
  const costTrajectory = {
    avgUSD: avg(costReportUSDValues),
    count: costReportUSDValues.length,
    lastCostAdvisorVerdict: null as string | null
  };
  return [
    ...renderSnapshotHeader(today),
    ...renderBacklogState(backlog, inFlight),
    ...renderInFlight(inFlight),
    ...renderTriagedTop(triagedTop),
    ...renderSpecs(specs, specsByStatus),
    ...renderGradeSignal(grades, recentScoreAverages),
    ...renderDecisions(recentDecisions),
    ...renderLoopTelemetry(iterationStatus, costTrajectory),
    ...renderUsageFooter()
  ];
}

export async function generateSnapshotContent(
  repoPath: string,
  { config = null, today }: { config?: Record<string, any> | null; today?: string } = {}
): Promise<string> {
  const data = await collectSnapshotData(repoPath, config);
  const date = today ?? new Date().toISOString().slice(0, 10);
  return buildSnapshotLines(data, date).join("\n");
}

export async function writeMemorySnapshot(
  repoPath: string,
  { config = null }: { config?: Record<string, any> | null } = {}
) {
  const data = await collectSnapshotData(repoPath, config);
  const today = new Date().toISOString().slice(0, 10);
  const lines = buildSnapshotLines(data, today);

  const target = resolveSnapshotPath(repoPath, config);
  await fs.mkdir(path.dirname(target), { recursive: true });
  await fs.writeFile(target, lines.join("\n"));
  return {
    artifactPath: target,
    counts: {
      backlog: data.backlog.length,
      specs: data.specs.length,
      grades: data.grades.length,
      decisions: data.decisions.length,
      costReports: data.costReportUSDValues.length
    }
  };
}

#!/usr/bin/env node
// PreToolUse hook script: validates plugin manifests before git commit and
// git push. Reads tool input from stdin. Exits 0 to allow, 2 to block.
//
// Two gates, escalating with the cost of the operation:
//   - git commit → manifest validation only (includes the version lockstep
//     check across plugin.json / marketplace.json / package.json). This is
//     the earliest point version drift can be caught. Recurring root cause of
//     drift: hand-editing one manifest and forgetting the others — always use
//     `bun run bump-version <v>` to bump all three in lockstep.
//   - git push → manifest validation + content-length + prettier format:check
//     (the full pre-push gate, mirrors CI). Also runs --check-generated in
//     warn-only mode to detect drift in auto-generated artifacts.

import { execFileSync } from "node:child_process";
import { fileURLToPath } from "node:url";
import path from "node:path";
import fs from "node:fs/promises";

const scriptDir = path.dirname(fileURLToPath(import.meta.url));

// --check-generated: generate expected snapshot content in-memory and diff
// against the on-disk file. Returns true if up to date, false if drift found.
async function checkGeneratedArtifacts(repoPath: string): Promise<boolean> {
  const { generateSnapshotContent } = await import("./lib/memory-snapshot.mjs");
  const { snapshotPath } = await import("./lib/paths.mjs");

  const target = snapshotPath(repoPath, null);
  const today = new Date().toISOString().slice(0, 10);

  let onDisk: string;
  try {
    onDisk = await fs.readFile(target, "utf8");
  } catch {
    process.stderr.write(`  warning: loop-snapshot.md not found at ${target}\n`);
    return false;
  }

  const expected = await generateSnapshotContent(repoPath, { today });
  if (onDisk === expected) return true;

  process.stderr.write(`  warning: loop-snapshot.md is out of date.\n`);
  process.stderr.write(`  Run: bun src/scripts/loop.mts snapshot-memory --repo <path>\n`);
  return false;
}

const args = process.argv.slice(2);
const checkGeneratedFlag = args.includes("--check-generated");

if (checkGeneratedFlag) {
  const repoPath = process.cwd();
  const upToDate = await checkGeneratedArtifacts(repoPath);
  if (upToDate) {
    process.stdout.write("✓ generated artifacts up to date\n");
    process.exit(0);
  } else {
    process.exit(1);
  }
}

let input = "";
for await (const chunk of process.stdin) input += chunk;

try {
  const data = JSON.parse(input);
  const cmd = data?.tool_input?.command || "";

  const isPush = /\bgit\s+push\b/.test(cmd);
  const isCommit = /\bgit\s+commit\b/.test(cmd);

  if (!isPush && !isCommit) {
    process.exit(0);
  }

  // Manifest validation (version lockstep + schema) runs on BOTH commit and
  // push, so version drift is blocked at the commit gate, not just at push.
  execFileSync("bun", [path.join(scriptDir, "validate-manifests.mts")], {
    stdio: ["ignore", "pipe", "pipe"],
    shell: process.platform === "win32"
  });

  if (isPush) {
    execFileSync("bun", [path.join(scriptDir, "validate-content-length.mts")], {
      stdio: ["ignore", "pipe", "pipe"],
      shell: process.platform === "win32"
    });
    // Prettier drift was undetected for several pushes; running format:check
    // here mirrors the CI gate so divergence blocks at git push, not in CI.
    execFileSync("bun", ["run", "format:check"], {
      stdio: ["ignore", "pipe", "pipe"],
      shell: process.platform === "win32"
    });
    // Warn-only: detect drift in auto-generated artifacts without blocking push.
    // A future slice will flip this to blocking (exit 2 on drift).
    const repoPath = process.cwd();
    await checkGeneratedArtifacts(repoPath).catch(() => {
      /* ignore errors so push is never blocked by a generator failure */
    });
  }
  process.exit(0);
} catch (e) {
  const err = e as { stderr?: string; message?: string };
  process.stderr.write(
    `Plugin validation failed — fix before git commit/push:\n${err.stderr || err.message}\n`
  );
  process.exit(2);
}

/** Extract the `- **FEAT-NNN**` bullet ids from the "Top triaged" section only. */
function topTriagedIds(content: string): string[] {
  const start = content.indexOf("## Top triaged (next to slice)");
  if (start === -1) return [];
  const rest = content.slice(start + 1);
  const end = rest.indexOf("\n## ");
  const section = end === -1 ? rest : rest.slice(0, end);
  return [...section.matchAll(/- \*\*(FEAT-\d+)\*\*/g)].map((m) => m[1]!);
}

async function addTriaged(repo: string, title: string, updates: Record<string, any>) {
  const f = await addFeature(repo, { title });
  await moveFeature(f.filePath, "triaged", { priority: "P2", category: "feature", ...updates });
  return f.id;
}

// AC-2: deferred FEAT excluded from Top-triaged when >=5 non-deferred exist
test("writeMemorySnapshot excludes deferred FEAT from Top triaged (>=5 non-deferred)", async () => {
  const repo = await tmpRepo("al-snapshot-deferred-excl-");
  const deferredId = await addTriaged(repo, "Blocked upstream", { priority: "P1", deferred: true });
  const liveIds: string[] = [];
  for (let i = 0; i < 5; i++) liveIds.push(await addTriaged(repo, `Live feature ${i}`, {}));

  const snapshot = await writeMemorySnapshot(repo);
  const content = await fs.readFile(snapshot.artifactPath, "utf8");
  const top = topTriagedIds(content);

  assert.ok(!top.includes(deferredId), `deferred ${deferredId} must not appear in Top triaged`);
  assert.equal(top.length, 5, "Top triaged should show 5 non-deferred entries");
  for (const id of top)
    assert.ok(liveIds.includes(id), `${id} should be a live (non-deferred) feature`);
});
